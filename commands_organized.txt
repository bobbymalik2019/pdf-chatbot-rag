3. HEALTH CHECKS / NETWORK TROUBLESHOOTING
======================================================================
# Chatbot FastAPI
curl.exe -i http://127.0.0.1:8000/health
# SSO and Chatbot Gateway base URLs
curl.exe -i http://127.0.0.1:3100/
curl.exe -i http://127.0.0.1:3000/

# Verify Windows hosts resolution; expected 127.0.0.1
ping -4 ssotest.charter26.com
ping -4 chatbottest.charter26.com
# Show hosts file (requires permissions for edits, not reading)
Get-Content "$env:SystemRoot\System32\drivers\etc\hosts"

# Check listeners for all important local ports
3000,3100,443,8000,8100,5433,11434 | ForEach-Object {
    Get-NetTCPConnection -LocalPort $_ -State Listen -ErrorAction SilentlyContinue |
        Select-Object LocalAddress,LocalPort,OwningProcess
}

# Identify process by PID from above
Get-Process -Id <PID>
# Stop an identified stale process only when safe
Stop-Process -Id <PID>

# HTTPS diagnostic: --ssl-no-revoke avoids Windows Schannel revocation check
# and should NOT be used as an application-level TLS bypass.
curl.exe --ssl-no-revoke -i https://ssotest.charter26.com/
curl.exe --ssl-no-revoke -i https://chatbottest.charter26.com/

# If host resolves but port 443 does not listen, start Caddy.
# If HTTPS returns 502, verify gateway port 3100 or 3000 is running.
# If HTTPS returns certificate error, inspect mkcert trust and certificate.
